\subsection{Contexte et motivation}

\begin{frame}{Contexte général}
  \begin{block}{Place des transferts de migrants dans l'économie sénégalaise}
    \begin{itemize}
      \item De 233 millions de dollars en 2000 à 2\,220 millions en 2017, soit de 166 à 1\,242 milliards de FCFA.
      \item 12,1\,\% du PIB : quatrième récepteur d'Afrique subsaharienne (ANSD/OIM, 2018).
      \item Des montants reçus dont l'ordre de grandeur avoisine le seuil de pauvreté monétaire annuel par tête.
      \pause
    \end{itemize}
  \end{block}
  \begin{alertblock}{Des privations infantiles d'une ampleur considérable}
    \begin{itemize}
      \item 50,7\,\% des enfants de 0 à 17 ans privés dans au moins 4 des 7 domaines de leur bien-être en 2018 (ANSD/UNICEF, 2024).
      \item Domaines : éducation, santé, nutrition, eau, assainissement, logement, protection.
      \item Des manques qui échappent largement aux mesures fondées sur le seul revenu du ménage.
    \end{itemize}
  \end{alertblock}
\end{frame}

\subsection{Problématique et objectifs}

\begin{frame}{Problématique}
  \begin{block}{Question de recherche}
    \justifying
    \large
    \textit{Dans quelle mesure les transferts de migrants réduisent-ils
    la pauvreté multidimensionnelle des enfants au Sénégal ?}
  \end{block}
  \medskip
  \begin{block}{Enjeu méthodologique}
    Les ménages bénéficiaires de transferts ne sont pas un échantillon aléatoire :
    biais de sélection observable et inobservable.
  \end{block}
  \begin{alertblock}{Réponse méthodologique}
    Combinaison PSM + double différence sur les données EHCVM~I et II (2018-2022).
  \end{alertblock}
\end{frame}

\begin{frame}{Objectifs et hypothèses}
  \small
  \begin{block}{Objectif général}
    Évaluer l'impact des transferts de migrants sur la pauvreté
    multidimensionnelle des enfants au Sénégal.
  \end{block}
  \pause
  \begin{block}{Objectifs spécifiques}
    \begin{itemize}
      \item Construire un indice de pauvreté multidimensionnelle adapté aux enfants sénégalais.
      \item Estimer l'impact des transferts sur cet indice et en analyser l'hétérogénéité.
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Hypothèses de recherche}
    \begin{enumerate}
      \item[H1] Les transferts réduisent significativement la pauvreté multidimensionnelle des enfants.
      \item[H2] L'effet est hétérogène selon les dimensions, le milieu, le genre du chef de ménage, l'âge de l'enfant et le montant reçu.
    \end{enumerate}
  \end{block}
\end{frame}

\subsection{Transferts de migrants et bien-être}

\begin{frame}{Canaux de transmission des transferts de migrants}
  \begin{block}{Théories sur les transferts}
    \begin{itemize}
      \item \textbf{Altruisme} (Lucas \& Stark, 1985) : les transferts décroissent avec le revenu du receveur.
      \item \textbf{Intérêt personnel et échange} (Rapoport \& Docquier, 2006) : préserver un héritage, rembourser la dette du départ.
      \item \textbf{Diversification du risque et assurance} (Stark \& Bloom, 1985) : les transferts assurent le ménage contre les chocs.
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Effets empiriques documentés}
    \begin{itemize}
      \item Réduction modeste de la pauvreté monétaire, atténuée par les coûts de transaction (Azam \& Gubert, 2006 ; Combes \& Ebeke, 2011).
      \item Scolarisation et investissement en capital humain (Cox Edwards \& Ureta, 2003 ; Yang, 2008).
      \item Effets fragiles sur la nutrition, conditionnels au maintien du lien avec le migrant (Davis \& Brazil, 2016).
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Pauvreté multidimensionnelle des enfants}

\begin{frame}{Approche multidimensionnelle de la pauvreté infantile}
  \small
  \begin{block}{Limites de l'approche monétaire}
    \begin{itemize}
      \item Le revenu/la consommation ne capturent pas toutes les privations des enfants.
      \item Les enfants ont des besoins spécifiques : nutrition, éducation, soins de santé.
    \end{itemize}
  \end{block}
  \begin{alertblock}{Trois exigences dégagées par la littérature}
    \begin{itemize}
      \item L'enfant individuel comme unité d'analyse, et non le ménage.
      \item Des indicateurs adaptés à l'âge de l'enfant.
      \item L'attention aux privations superposées plutôt qu'à chaque manque isolé.
    \end{itemize}
  \end{alertblock}
  \begin{block}{Lacune dans la littérature}
    Peu d'études combinent PSM et double différence pour identifier l'effet
    des transferts sur la pauvreté multidimensionnelle des enfants au Sénégal.
  \end{block}
\end{frame}

\subsection{Données EHCVM}

\begin{frame}{Données EHCVM I et II}
  \begin{block}{Source et période}
    \begin{itemize}
      \item \textbf{EHCVM I (2018-2019)} : première vague harmonisée UEMOA au Sénégal.
      \item \textbf{EHCVM II (2021-2022)} : deuxième vague permettant de constituer un panel.
      \item \textbf{Panel vrai} : 6\,127 ménages suivis entre les deux vagues.
      \item \textbf{Traitement} : ménages bénéficiaires en 2018 d'un transfert
            de migrant issu du ménage, comparés aux non-bénéficiaires.
    \end{itemize}
  \end{block}
  \begin{block}{Population cible et unité d'analyse}
    Enfants de 0 à 17 ans suivis individuellement d'une vague à l'autre, soit
    17\,786 enfants : 2\,638 dans un ménage bénéficiaire en 2018 et
    15\,148 dans un ménage non bénéficiaire. Le score de propension et
    l'appariement sont estimés à ce niveau.
  \end{block}
  \begin{alertblock}{Covariables du modèle de propension}
    Ménage : âge, genre, niveau d'éducation, statut matrimonial et catégorie
    socioprofessionnelle du chef ; taille du ménage ; dépense par tête ;
    milieu de résidence ; région. Enfant : genre et âge.
  \end{alertblock}
\end{frame}

\subsection{Mesure de la pauvreté multidimensionnelle}

\begin{frame}{MODA en 7 dimensions et 3 groupes d'âge}
  \begin{block}{Méthodologie MODA (ANSD/UNICEF)}
    Sept dimensions du bien-être de l'enfant : assainissement, eau, logement,
    nutrition, santé, protection et éducation. Quatorze indicateurs, déclinés
    selon le groupe d'âge (0-4, 5-14, 15-17 ans). Le détail figure en annexe.
  \end{block}
  \begin{alertblock}{Double seuil MODA}
    Union intra-dimension (privé si privé dans au moins un indicateur) ;
    seuil inter-dimensionnel $k = 4$ (pauvre si privé dans $\geq 4$ dimensions sur 7).
  \end{alertblock}
  \begin{center}
    \hyperlink{matrice-nmoda}{\beamergotobutton{Voir la matrice MODA complète}}
  \end{center}
\end{frame}

\begin{frame}{Le traitement, les transferts de migrants en pratique}
  \footnotesize
  \begin{block}{Motifs déclarés des transferts étrangers (EHCVM I / II, \% des transferts)}
    \smallskip
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}lcc@{}}
      \toprule
      Motif principal & 2018 & 2021 \\
      \midrule
      Soutien courant            & 66,7\,\% & 68,5\,\% \\
      Fêtes et évènements        & 11,3\,\% & 9,3\,\% \\
      Santé, maladie             & 4,6\,\%  & 6,5\,\% \\
      Scolarité, éducation       & 3,5\,\%  & 3,7\,\% \\
      Autres motifs              & 13,9\,\% & 12,0\,\% \\
      \bottomrule
    \end{tabular*}
  \end{block}
  \begin{alertblock}{Montants annuels reçus par ménage bénéficiaire}
    Médiane $\approx$ 600\,000 FCFA/an, troisième quartile à 1\,280\,000 :
    une ressource loin d'être marginale, très inégalement répartie.
  \end{alertblock}
  \begin{block}{Implication méthodologique}
    Deux tiers des transferts relèvent du soutien courant, non affecté :
    l'argent étant fongible, on mesure l'effet sur les privations
    effectives des enfants plutôt que sur l'affectation déclarée des fonds.
  \end{block}
\end{frame}

\subsection{Stratégie d'identification}

\begin{frame}{Stratégie d'identification, le problème (1/2)}
  \small
  \begin{alertblock}{Traitement, statut défini en 2018}
    Traités : ménages recevant en 2018 un transfert d'un expéditeur vivant hors
    du Sénégal et ayant déjà vécu dans le ménage. Témoins : ménages non
    bénéficiaires. Le statut, figé en 2018, ne peut pas dépendre de l'évolution
    ultérieure des privations.
  \end{alertblock}

  \begin{block}{Problème du biais de sélection}
    L'effet moyen du traitement sur les traités (ATT) :
    \begin{equation*}
      \tau_{ATT} = \mathbb{E}[Y_i(1) - Y_i(0) \mid D_i = 1]
    \end{equation*}
    $Y_i(0)$ (contrefactuel) est inobservable $\Rightarrow$ nécessité d'un estimateur adapté.
  \end{block}
\end{frame}

\begin{frame}{Stratégie d'identification, l'estimateur (2/2)}
  \small
  \begin{block}{Étape 1 - Appariement par score de propension (PSM)}
    Estimation du score de propension par logit :
    \begin{equation*}
      p(X_i) = P(D_i = 1 \mid X_i) = \frac{\exp(X_i'\beta)}{1+\exp(X_i'\beta)}
    \end{equation*}
    Appariement par $k$-plus proches voisins ($k=4$) et vérification du support commun.
  \end{block}

  \begin{alertblock}{Étape 2 - Double différence (DD) après appariement}
    \begin{equation*}
      \hat{\tau}_{PSM\text{-}DD}
      = \frac{1}{n_T}\sum_{i\in\mathcal{T}}
        \Bigl[\underbrace{Y_{i,t_2}-Y_{i,t_1}}_{\Delta Y_i}
        - \sum_{j\in\mathcal{C}} w_{ij}\,\Delta Y_j\Bigr]
    \end{equation*}
    Contrôle simultané des biais observables (PSM) et des effets fixes inobservés (DD).
  \end{alertblock}
\end{frame}

\subsection{Impact des transferts}

\begin{frame}{Impact des transferts sur la pauvreté multidimensionnelle}
  \footnotesize
  \begin{table}
    \centering
    \caption{Effet moyen sur les traités (ATT), pauvreté MODA}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{@{}lccc@{}}
      \toprule
      Spécification & ATT & Écart-type & Significativité \\
      \midrule
      DD brute (sans appariement) & $0{,}035$ & $(0{,}025)$ & \\
      PSM-DD, k-NN ($k=4$)        & $0{,}064$ & $(0{,}029)$ & $^{**}$ \\
      PSM-DD, noyau Epanechnikov  & $0{,}061$ & $(0{,}032)$ & $^{*}$ \\
      PSM-DD, caliper             & $0{,}050$ & $(0{,}028)$ & $^{*}$ \\
      \bottomrule
    \end{tabular}
    \par\smallskip
    {\scriptsize $^{*}p<0{,}10$, $^{**}p<0{,}05$.}
  \end{table}
  \begin{alertblock}{Résultat central, pas de réduction}
    Sur 2018-2021, la pauvreté multidimensionnelle des enfants des ménages
    bénéficiaires recule plus lentement que celle des témoins appariés
    ($+6{,}4$ pp, significatif à 5\,\%), les trois
    appariements concordant (0,050 à 0,064). H1 n'est pas validée.
  \end{alertblock}
  \begin{block}{Lecture}
    L'effet n'est pas diffus : il n'est établi que sur la nutrition, chez les
    0-4 ans et sur les plus faibles montants reçus.
  \end{block}

\subsection{Tests de robustesse}

\begin{frame}{Tests de robustesse}
  \begin{block}{Sensibilité des résultats}
    \begin{itemize}
      \item Méthodes d'appariement (k-NN, noyau, caliper) : effet de
            $0{,}050$ à $0{,}064$, k-NN significatif à 5\,\%, noyau et
            caliper à 10\,\%.
      \item Seuil $k$, testé de 1 à 7 : ampleur en courbe de cloche,
            maximale à $k=4$ et significative seulement à $k=2$, $k=3$ et
            $k=4$ : l'effet se joue au milieu de la distribution des
            privations, pas dans son noyau.
      \item Sans appariement, sur tous les enfants suivis :
            $0{,}035$, non significatif, même signe et ampleur atténuée,
            l'avantage initial des bénéficiaires n'étant pas corrigé.
    \end{itemize}
  \end{block}
  \begin{alertblock}{Définition alternative, les bénéficiaires stables}
    Chez les bénéficiaires des deux vagues, la DD tombe à $0{,}022$ et le
    PSM-DD à $-0{,}008$, tous deux non significatifs. L'exposition de ces
    ménages, déjà en cours en 2018, relève d'une autre question que le design
    principal.
  \end{alertblock}
\end{frame}

\section[Conclusion]{Conclusion et recommandations}
% ============================================================

\subsection{Conclusion générale}

\begin{frame}{Conclusion générale}
  \begin{block}{Principaux apports}
    \begin{enumerate}
      \item Mesure de la pauvreté multidimensionnelle des enfants sur deux
            vagues EHCVM selon l'approche MODA (7 dimensions, $k=4$).
      \pause
      \item L'incidence MODA recule de 63,5\,\% (2018) à 57,8\,\% (2021) :
            amélioration réelle mais la pauvreté reste massive.
      \pause
      \item Les transferts ne réduisent pas, à trois ans, la
            pauvreté multidimensionnelle des enfants : l'effet est de
            $+0{,}064$, significatif à 5\,\%, de signe défavorable et stable aux
            trois appariements.
      \pause
      \item Il n'est établi que sur la nutrition, chez les 0-4 ans et sur les
            plus faibles montants reçus, là où l'apport monétaire est trop
            faible pour peser sur les conditions de vie de l'enfant.
    \end{enumerate}
  \end{block}
\end{frame}

\subsection{Recommandations}

\begin{frame}{Recommandations de politique économique}
  \begin{block}{Cibler là où l'écart se concentre}
    \begin{itemize}
      \item Sécurité alimentaire des enfants en priorité dans les zones d'émigration.
      \item Appui ciblé aux enfants de 0 à 4 ans des ménages bénéficiaires.
      \item Accompagner en priorité les ménages recevant les plus faibles montants.
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Améliorer les conditions de conversion des transferts}
    \begin{itemize}
      \item Réduire les coûts de transaction des envois formels.
      \item Compléter les transferts par l'offre publique de services de base (eau, assainissement, santé).
      \item Orienter les usages vers le capital humain et renforcer le suivi statistique des transferts.
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Limites et perspectives}

\begin{frame}{Limites et perspectives de recherche}
  \small
  \begin{block}{Choix de mesure assumé}
    Nutrition et santé reposent sur les seuls indicateurs comparables entre les
    deux vagues. La comparabilité dans le temps conditionne la double
    différence : un indicateur redéfini entre 2018 et 2021 produirait un effet
    sans contenu causal.
  \end{block}
  \begin{alertblock}{Limites de l'étude}
    \begin{itemize}
      \item Hypothèse de tendances parallèles non testable
            directement, deux vagues seulement étant disponibles ;
            l'appariement sur les niveaux initiaux la rend plus crédible.
      \item Vieillissement des enfants entre les deux enquêtes, chaque
            enfant étant évalué sur la grille de son âge courant à chaque
            vague.
    \end{itemize}
  \end{alertblock}
  \pause
  \begin{block}{Perspectives}
    \begin{itemize}
      \item Analyser les mécanismes de transmission (dépenses en éducation, santé, logement).
      \item Étendre l'analyse aux autres pays de l'UEMOA couverts par l'EHCVM.
      \item Intégrer la durée d'exposition aux transferts et leurs usages effectifs.
    \end{itemize}
  \end{block}
\end{frame}
