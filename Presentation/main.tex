\subsection{Contexte et motivation}

\begin{frame}{Contexte général}
  \begin{block}{Place des transferts de migrants dans l'économie sénégalaise}
    \begin{itemize}
      \item 2\,220 millions de dollars reçus en 2017, soit 1\,242 milliards de FCFA et 12,1\,\% du PIB.
      \item Première source de devises, devançant l'aide publique au développement et les IDE.
      \item Europe 65\,\% des envois, Afrique 30\,\% : une migration autant régionale qu'intercontinentale.
      \item Plus de 3 millions de Sénégalais vivent à l'étranger (diaspora).
      \pause
    \end{itemize}
  \end{block}
  \begin{alertblock}{Pauvreté infantile : un défi persistant}
    \begin{itemize}
      \item Près de 65\,\% des enfants souffrent de privations multiples (MODA, EHCVM I).
      \item Privations cumulées : éducation, santé, eau, assainissement, logement, nutrition, protection.
      \item La pauvreté monétaire ne suffit pas à capturer toutes ces dimensions.
    \end{itemize}
  \end{alertblock}
\end{frame}

\subsection{Problématique et objectifs}

\begin{frame}{Problématique}
  \begin{block}{Question de recherche}
    \justifying
    \large
    \textit{Dans quelle mesure les transferts de migrants réduisent-ils
    la pauvreté multidimensionnelle des enfants au Sénégal ?}
  \end{block}
  \medskip
  \begin{block}{Enjeu méthodologique}
    Les ménages bénéficiaires de transferts ne sont pas un échantillon aléatoire :
    biais de sélection observable et inobservable.
  \end{block}
  \begin{alertblock}{Réponse méthodologique}
    Combinaison PSM + double différence sur les données EHCVM~I et II (2018-2022).
  \end{alertblock}
\end{frame}

\begin{frame}{Objectifs et hypothèses}
  \small
  \begin{block}{Objectif général}
    Évaluer l'impact causal des transferts de migrants sur la pauvreté
    multidimensionnelle des enfants au Sénégal.
  \end{block}
  \pause
  \begin{block}{Objectifs spécifiques}
    \begin{itemize}
      \item Mesurer la pauvreté multidimensionnelle des enfants selon l'approche MODA (EHCVM I \& II).
      \item Analyser son évolution entre 2018-2019 et 2021-2022.
      \item Estimer l'effet causal des transferts par PSM-Double différence.
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Hypothèses de recherche}
    \begin{enumerate}
      \item[H1] Les transferts réduisent significativement la pauvreté multidimensionnelle des enfants.
      \item[H2] L'effet est hétérogène selon les dimensions, le milieu, le sexe de l'enfant et le montant reçu.
      \item[H3] Le PSM-DD contrôle efficacement les biais de sélection observés et inobservés.
    \end{enumerate}
  \end{block}
\end{frame}

\section[Revue]{Revue de la littérature}

\subsection{Transferts de migrants et bien-être}

\begin{frame}{Transferts de migrants : canaux de transmission}
  \begin{block}{Théories sur les transferts}
    \begin{itemize}
      \item \textbf{Altruisme} (Lucas \& Stark, 1985) : les migrants soutiennent leur famille par solidarité.
      \item \textbf{Contrat familial implicite} : les transferts compensent l'investissement consenti dans la migration.
      \item \textbf{Diversification des risques} : le ménage envoie un membre migrer pour sécuriser les revenus.
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Effets empiriques documentés}
    \begin{itemize}
      \item Réduction de la pauvreté monétaire (Adams \& Page, 2005 ; Acosta et al., 2008).
      \item Amélioration de la scolarisation et de la santé des enfants (Yang, 2008).
      \item Effets plus marqués en milieu rural et pour les ménages les plus pauvres.
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Pauvreté multidimensionnelle des enfants}

\begin{frame}{Approche multidimensionnelle de la pauvreté infantile}
  \small
  \begin{block}{Limites de l'approche monétaire}
    \begin{itemize}
      \item Le revenu/la consommation ne capturent pas toutes les privations des enfants.
      \item Les enfants ont des besoins spécifiques : nutrition, éducation, soins de santé.
    \end{itemize}
  \end{block}
  \begin{alertblock}{L'approche Alkire-Foster (2011)}
    \begin{itemize}
      \item Identification des privations dans plusieurs dimensions simultanément.
      \item Double seuil : intra-dimensionnel (seuil de privation) et inter-dimensionnel (seuil de pauvreté $k$).
      \item Mesure $M_0 = H \times A$ : incidence $\times$ intensité.
    \end{itemize}
  \end{alertblock}
  \begin{block}{Lacune dans la littérature}
    Peu d'études combinent PSM et double différence pour identifier l'effet causal
    des transferts sur la pauvreté multidimensionnelle des enfants au Sénégal.
  \end{block}
\end{frame}

\subsection{Mesure de la pauvreté multidimensionnelle}

\begin{frame}{MODA : 7 dimensions, 3 groupes d'âge}
  \begin{block}{Méthodologie MODA (ANSD/UNICEF)}
    Sept dimensions du bien-être de l'enfant (assainissement, eau, logement,
    nutrition, santé, protection, éducation), déclinées en indicateurs
    applicables selon le groupe d'âge (0-4, 5-14, 15-17 ans). Le détail complet
    (indicateurs, modules sources, seuils de privation) figure en annexe.
  \end{block}
  \begin{alertblock}{Double seuil MODA}
    Union intra-dimension (privé si privé dans au moins un indicateur) ;
    seuil inter-dimensionnel $k = 4$ (pauvre si privé dans $\geq 4$ dimensions sur 7).
  \end{alertblock}
  \begin{center}
    \hyperlink{matrice-nmoda}{\beamergotobutton{Voir la matrice MODA complète}}
  \end{center}
\end{frame}

\begin{frame}{Le traitement : les transferts de migrants en pratique}
  \footnotesize
  \begin{block}{Motifs déclarés des transferts étrangers (EHCVM I / II, \% des transferts)}
    \centering\smallskip
    \begin{tabular}{lcc}
      \toprule
      Motif principal & 2018 & 2021 \\
      \midrule
      Soutien courant            & 66,7\,\% & 68,5\,\% \\
      Fête, évènements           & 11,3\,\% & 9,3\,\% \\
      Santé, maladie             & 4,6\,\%  & 6,5\,\% \\
      Scolarité, éducation       & 3,5\,\%  & 3,7\,\% \\
      Autres motifs              & 13,9\,\% & 12,0\,\% \\
      \bottomrule
    \end{tabular}
  \end{block}
  \begin{alertblock}{Montants annuels reçus par ménage bénéficiaire}
    Médiane $\approx$ 300\,000 FCFA/an ; moyenne $\approx$ 900\,000 FCFA/an
    (forte dispersion) : une ressource loin d'être marginale.
  \end{alertblock}
  \begin{block}{Implication méthodologique}
    Deux tiers des transferts relèvent du soutien courant, non affecté :
    l'argent étant fongible, on mesure l'effet sur les privations
    effectives des enfants plutôt que sur l'affectation déclarée des fonds.
  \end{block}
\end{frame}

\subsection{Stratégie d'identification}

\begin{frame}{PSM + Double différence : principe (1/2)}
  \small
  \begin{alertblock}{Traitement : statut défini en 2018}
    Traités = ménages recevant en 2018 un transfert d'un expéditeur résidant
    hors du Sénégal et ayant déjà vécu dans le ménage
    (migration issue du ménage) ; témoins = ménages non bénéficiaires. Le
    statut, fixé à la période de base, ne peut pas être affecté par
    l'évolution ultérieure des privations.
  \end{alertblock}

  \begin{block}{Problème du biais de sélection}
    L'effet moyen du traitement sur les traités (ATT) :
    \begin{equation*}
      \tau_{ATT} = \mathbb{E}[Y_i(1) - Y_i(0) \mid D_i = 1]
    \end{equation*}
    $Y_i(0)$ (contrefactuel) est inobservable $\Rightarrow$ nécessité d'un estimateur adapté.
  \end{block}
\end{frame}

\begin{frame}{PSM + Double différence : principe (2/2)}
  \small
  \begin{block}{Étape 1 - Appariement par score de propension (PSM)}
    Estimation du score de propension par probit :
    \begin{equation*}
      p(X_i) = P(D_i = 1 \mid X_i) = \Phi(X_i'\beta)
    \end{equation*}
    Appariement par $k$-plus proches voisins ($k=4$) et vérification du support commun.
  \end{block}

  \begin{alertblock}{Étape 2 - Double différence (DD) après appariement}
    \begin{equation*}
      \hat{\tau}_{PSM\text{-}DD}
      = \frac{1}{n_T}\sum_{i\in\mathcal{T}}
        \Bigl[\underbrace{Y_{i,t_2}-Y_{i,t_1}}_{\Delta Y_i}
        - \sum_{j\in\mathcal{C}} w_{ij}\,\Delta Y_j\Bigr]
    \end{equation*}
    Contrôle simultané des biais observables (PSM) et des effets fixes inobservés (DD).
  \end{alertblock}
\end{frame}

\subsection{Données EHCVM}

\begin{frame}{Données : EHCVM I et II}
  \begin{block}{Source et période}
    \begin{itemize}
      \item \textbf{EHCVM I (2018-2019)} : première vague harmonisée UEMOA au Sénégal.
      \item \textbf{EHCVM II (2021-2022)} : deuxième vague permettant de constituer un panel.
      \item \textbf{Panel vrai} : 6\,127 ménages suivis entre les deux vagues.
      \item \textbf{Traitement} : ménages bénéficiaires en 2018 d'un transfert
            de migrant issu du ménage, comparés aux non-bénéficiaires.
    \end{itemize}
  \end{block}
  \begin{block}{Population cible et unité d'analyse}
    Enfants de 0 à 17 ans suivis individuellement d'une vague à l'autre par
    l'identifiant préchargé du questionnaire 2021, soit 17\,786 enfants. Le
    score de propension et l'appariement sont estimés à ce niveau.
  \end{block}
  \begin{alertblock}{Covariables du modèle de propension}
    Ménage : âge, sexe, niveau d'éducation, statut matrimonial et catégorie
    socioprofessionnelle du chef ; taille du ménage ; dépense par tête ;
    milieu de résidence ; région. Enfant : sexe et âge.
  \end{alertblock}
\end{frame}

\begin{frame}{Pauvreté multidimensionnelle des enfants : état des lieux}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{Résultats MODA ($k=4$)}

      \midrule
      Garçons   & 66,2 & 50,5 & 59,7 & 43,5 & $-0{,}5$ \\

    \end{tabular}
  \end{table}
  \begin{alertblock}{L'écart se referme chez les tout-petits et chez les filles}
    Chez les 0-4 ans, l'avantage des bénéficiaires passe de 21,4 à
    13,4 points ; chez les 5-14 ans il ne bouge pas. Le même contraste
    s'observe selon le sexe : $+5{,}0$ points chez les filles contre
    $-0{,}5$ chez les garçons. Le descriptif annonce le résultat causal.
  \end{alertblock}

    Les deux clivages se cumulent : chez les filles de 0 à 4 ans la double
    différence atteint $+12{,}5$ points, contre $+3{,}9$ chez les garçons du
    même âge. La protection joue par ailleurs en sens inverse à tout âge, et
    d'autant plus que l'enfant est grand ($+17{,}4$ points chez les
    15-17 ans) : à cet âge l'indicateur ne mesure que la séparation parentale.
  \end{block}
  \begin{center}
    \hyperlink{distrib-dimensions}{\beamergotobutton{Distribution selon le seuil}}
  \end{center}
\end{frame}

\subsection{Impact des transferts}

\begin{frame}{Impact des transferts sur la pauvreté multidimensionnelle}
  \footnotesize
  \begin{table}
    \centering
    \caption{Effet moyen sur les traités (ATT), pauvreté MODA}
    \renewcommand{\arraystretch}{1.1}
    \begin{tabular}{lc}
      \toprule
      Spécification & ATT \\
      \midrule
      DD brute (sans appariement)      & \makecell{$0{,}032$\\(0,023)} \\
      PSM-DD, k-NN ($k=4$)             & \makecell{$0{,}058^{*}$\\(0,030)} \\
      PSM-DD, noyau Epanechnikov       & \makecell{$0{,}055^{*}$\\(0,031)} \\
      PSM-DD, caliper                  & \makecell{$0{,}047^{*}$\\(0,027)} \\
      \bottomrule
    \end{tabular}
    {\scriptsize Écarts-types entre parenthèses. $^{*}p<0{,}10$.}
  \end{table}
  \begin{alertblock}{Résultat central : pas de réduction, présomption d'effet défavorable}
    Sur 2018-2021, la pauvreté multidimensionnelle des enfants des ménages
    bénéficiaires recule plus lentement que celle des témoins appariés
    ($+5{,}8$ pp). Significatif à 10\,\%, pas à 5\,\% : H1 est rejetée,
    l'effet inverse est suggéré sans être démontré. Les trois appariements
    concordent (0,047 à 0,058).
  \end{alertblock}
  \begin{block}{Lecture}
    L'effet n'est pas diffus : nutrition, 0-4 ans, petits montants. Les transferts
    étant indissociables du départ d'un adulte, le coût de l'absence
    est le canal le plus plausible.
  \end{block}
\end{frame}

\section[Robustesse]{Robustesse et discussion}

\subsection{Tests de robustesse}

\begin{frame}{Tests de robustesse}
  \begin{block}{Sensibilité des résultats}
    \begin{itemize}
      \item Méthodes d'appariement (k-NN, noyau, caliper) : ATT de
            $0{,}047$ à $0{,}058$, tous significatifs à 10\,\%.
      \item Seuil $k$, testé de 1 à 7 : ampleur en courbe de cloche,
            maximale à $k=4$ et significative seulement à $k=3$ et $k=4$ --
            l'effet se joue au milieu de la distribution des privations,
            pas dans son noyau.
      \item Unité d'appariement : sans appariement individuel, sur tous les
            enfants du panel de ménages, ATT $= 0{,}032$ ($p=0{,}167$) --
            même signe, ampleur atténuée.
      \item Test placebo (200 assignations aléatoires) : distribution
            centrée sur zéro, ATT réel au 99,0\textsuperscript{e} centile.
      \item Attrition non corrélée au traitement ($p=0{,}101$).
    \end{itemize}
  \end{block}
  \begin{alertblock}{Définition alternative : les bénéficiaires durables}
    Chez les ménages bénéficiaires aux deux vagues, l'effet tombe à $0{,}023$
    et perd sa significativité ($p=0{,}474$) : l'écart de trajectoire ne se
    retrouve pas chez les bénéficiaires de longue date. Mais le traitement y
    est déjà en cours à $t=0$, donc pas de vraie période pré-traitement.
  \end{alertblock}
\end{frame}

\section[Conclusion]{Conclusion et recommandations}
% ============================================================

\subsection{Conclusion générale}

\begin{frame}{Conclusion générale}
  \begin{block}{Principaux apports}
    \begin{enumerate}
      \item Mesure de la pauvreté multidimensionnelle des enfants sur deux
            vagues EHCVM selon l'approche MODA (7 dimensions, $k=4$).
      \pause
      \item L'incidence MODA recule de 64,6\,\% (2018) à 57,3\,\% (2021) :
            amélioration réelle mais la pauvreté reste massive.
      \pause
      \item Recevoir des transferts ne réduit pas, à trois ans, la
            pauvreté multidimensionnelle des enfants : l'effet est de
            $+0{,}058$ ($p=0{,}052$), de signe défavorable et stable aux
            trois appariements.
      \pause
      \item L'effet se concentre sur la nutrition, les 0-4 ans et les plus
            faibles montants reçus : le coût du départ de l'adulte pèse sur
            les plus dépendants, et l'argent ne le compense pas.
    \end{enumerate}
  \end{block}
\end{frame}

\subsection{Recommandations}

\begin{frame}{Recommandations de politique économique}
  \begin{block}{Faciliter et orienter les transferts}
    \begin{itemize}
      \item Réduire les coûts de transaction des envois de fonds formels.
      \item Encourager le fléchage des transferts vers l'éducation et la santé des enfants.
      \item Développer des produits d'épargne liés aux transferts pour les ménages ruraux.
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Renforcer les politiques sociales complémentaires}
    \begin{itemize}
      \item Accompagner les ménages nouvellement bénéficiaires dans la durée
            (inclusion financière, appui à l'investissement dans l'habitat).
      \item Améliorer l'offre de services publics (écoles, centres de santé) dans les zones d'émigration.
      \item Renforcer les systèmes de collecte de données sur les migrations et les transferts.
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Limites et perspectives}

\begin{frame}{Limites et perspectives de recherche}
  \small
  \begin{block}{Choix de mesure assumé}
    Nutrition et santé sont mesurées par les seuls indicateurs strictement
    comparables entre les deux vagues (insécurité alimentaire vécue, accès aux
    soins). La comparabilité inter-temporelle est la condition de la double
    différence : un indicateur changeant de définition entre 2018 et 2021
    produirait un effet mesuré sans contenu causal.
  \end{block}
  \begin{alertblock}{Limites de l'étude}
    \begin{itemize}
      \item Panel non cylindré : attrition sélective (ménages urbains et aisés).
      \item L'estimation capte l'effet conjoint du départ du migrant et des transferts ;
            durée d'exposition variable au sein des bénéficiaires (effet de court terme).
      \item Effet significatif à 10\,\% seulement : intervalle de confiance
            à 95\,\% large ($[-0{,}1 ; 11{,}6]$ pp), présomption plutôt que
            démonstration.
    \end{itemize}
  \end{alertblock}
  \pause
  \begin{block}{Perspectives}
    \begin{itemize}
      \item Analyser les mécanismes de transmission (dépenses en éducation, santé, logement).
      \item Étendre l'analyse aux autres pays de l'UEMOA couverts par l'EHCVM.
      \item Intégrer la dimension genre de l'enfant et du chef de ménage.
    \end{itemize}
  \end{block}
\end{frame}
