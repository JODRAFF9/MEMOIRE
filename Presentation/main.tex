% -- Bandeau en ruban ----------------------------------------
% [#1] modele de titre. Par defaut r, le ruban a replis lateraux :
% l'intitule est porte par une banderole aux extremites entaillees en
% chevron, que flanquent deux replis dans la teinte assombrie. Les
% autres modeles restent disponibles a la demande : o onglet arrondi,
% f titre souligne d'un filet, p titre precede d'une pastille, b barre
% pleine largeur.
% #2 teinte, #3 largeur, #4 intitule, #5 corps.
% La largeur est donnee en fraction de \linewidth : le bandeau s'adapte
% ainsi a la colonne qui le contient.
\newcommand{\bandeau}[5][r]{%
  \begin{tikzpicture}
    \node[relief, fill=#2!8, rounded corners=4pt, text width=#3, align=justify,
          inner sep=8pt, font=\footnotesize, anchor=north west] (corps) at (0,0) {#5};
    % \ifstrequal vient d'etoolbox, que beamer charge deja
    \ifstrequal{#1}{b}{%
      % barre pleine largeur : l'intitule couvre tout le panneau
      \node[fill=#2, rounded corners=4pt, text=white, minimum height=0.6cm,
            text width=#3, inner sep=8pt, align=left,
            font=\footnotesize\bfseries, anchor=south west]
        (tete) at ([yshift=1pt]corps.north west) {#4};
      \draw[#2!62!black, line width=1.6pt, line cap=round]
        ([xshift=1pt]tete.south west) -- ([xshift=-1pt]tete.south east);}{%
      \ifstrequal{#1}{f}{%
        % intitule nu, souligne d'un filet courant sur toute la largeur
        \node[text=#2!70!black, minimum height=0.5cm, inner xsep=8pt,
              inner ysep=3pt, font=\footnotesize\bfseries, anchor=south west]
          (tete) at ([yshift=3pt]corps.north west) {#4};
        \draw[#2!62!black, line width=1.6pt, line cap=round]
          ([yshift=1pt]corps.north west) -- ([yshift=1pt]corps.north east);}{%
        \ifstrequal{#1}{p}{%
          % pastille pleine puis intitule dans la teinte
          \node[text=#2!70!black, minimum height=0.5cm, inner ysep=3pt,
                inner xsep=0pt, font=\footnotesize\bfseries, anchor=south west]
            (tete) at ([xshift=22pt, yshift=4pt]corps.north west) {#4};
          \fill[#2] ([xshift=-14pt]tete.west) circle (4.6pt);}{%
        \ifstrequal{#1}{o}{%
          % onglet plein aux quatre coins arrondis
          \node[fill=#2, rounded corners=4pt, text=white, minimum height=0.6cm,
                inner xsep=12pt, font=\footnotesize\bfseries, anchor=south west]
            (tete) at ([yshift=1pt]corps.north west) {#4};
          \draw[#2!62!black, line width=1.6pt, line cap=round]
            ([xshift=1pt]tete.south west) -- ([xshift=-1pt]tete.south east);}{%
          % ruban. Un premier noeud, invisible, ne sert qu'a mesurer
          % l'intitule ; la banderole et les replis sont ensuite traces,
          % puis le texte est repose par-dessus.
          \coordinate (ancre) at ([xshift=0.34cm]corps.north west);
          \node[minimum height=0.64cm, inner xsep=15pt, opacity=0,
                font=\footnotesize\bfseries, anchor=south west]
            (tete) at (ancre) {#4};
          % replis : deux triangles rectangles. Le cote vertical est
          % partage avec l'extremite de la banderole, la base repose sur
          % le bas de la banderole et l'angle droit est a l'interieur.
          \fill[#2!58!black] (tete.north west) -- (tete.south west)
            -- ([xshift=-0.34cm]tete.south west) -- cycle;
          \fill[#2!58!black] (tete.north east) -- (tete.south east)
            -- ([xshift=0.34cm]tete.south east) -- cycle;
          % banderole a extremites droites, posee par-dessus les replis.
          % Un degrade vertical et un filet clair sur l'arete haute lui
          % donnent du relief.
          \shade[top color=#2!72, bottom color=#2]
            (tete.north west) rectangle (tete.south east);
          \draw[white, opacity=0.30, line width=0.7pt]
            ([yshift=-0.4pt]tete.north west) -- ([yshift=-0.4pt]tete.north east);
          \node[minimum height=0.64cm, inner xsep=15pt, text=white,
                font=\footnotesize\bfseries, anchor=south west]
            at (ancre) {\ombretexte{#4}};}}}}
  \end{tikzpicture}}

\newcommand{\semis}[5]{%
  \begin{tikzpicture}[remember picture, overlay]
    \begin{scope}
      \clip ([xshift=#2 cm]current page.#1) rectangle ([yshift=#3 cm]current page.#1);
      \pgfmathsetseed{#5}
      \foreach \couleur/\nb in {presvert/130, presvertc/95, presorange/40} {
        \foreach \i in {1,...,\nb} {
          \pgfmathsetmacro{\dx}{rnd*(#2)}
          \pgfmathsetmacro{\dy}{rnd*(#3)}
          \pgfmathsetmacro{\rayon}{0.035 + rnd*0.10}
          \fill[\couleur, opacity=#4]
            ([xshift=\dx cm, yshift=\dy cm]current page.#1) circle (\rayon cm);
        }
      }
    \end{scope}
  \end{tikzpicture}}

\subsection{Problématique et objectifs}

\begin{frame}{Problématique}
  \bandeau{presvert}{0.93\linewidth}{Question de recherche}{%
    \justifying \large \textit{Dans quelle mesure, les transferts de migrants influencent-ils la pauvreté multidimensionnelle des enfants au Sénégal ?}}
\end{frame}

\begin{frame}{Objectifs et hypothèses}
  \small
  \bandeau{presvert}{0.93\linewidth}{Objectif général}{%
    {\color{presorange}\faBullseye}~Évaluer l'impact des transferts de migrants sur la pauvreté multidimensionnelle des enfants au Sénégal.}
  \begin{block}{Objectifs spécifiques}
    \begin{itemize}
      \item[{\color{presvert}\faBullseye}] Construire un indice de pauvreté
            multidimensionnelle adapté aux enfants sénégalais.
      \item[{\color{presvert}\faBullseye}] Estimer l'impact des transferts sur
            cet indice et en analyser l'hétérogénéité.
    \end{itemize}
  \end{block}
  \begin{block}{Hypothèses de recherche}
    \begin{enumerate}
      \item[H1] Les transferts réduisent significativement la pauvreté multidimensionnelle des enfants.
      \item[H2] L'effet est hétérogène selon les dimensions, le milieu, le genre du chef de ménage, l'âge de l'enfant et le montant reçu.
    \end{enumerate}
  \end{block}
\end{frame}

      = \frac{1}{n_T}\sum_{i\in\mathcal{T}}
        \Bigl[\Delta Y_i - \sum_{j\in\mathcal{C}} w_{ij}\,\Delta Y_j\Bigr]
    \end{equation*}

    Les deux limites se compensent : l'appariement rend les groupes comparables
    sur l'observable, la double différence neutralise l'inobservable stable
    (Heckman \emph{et al.}, 1997).
  \end{alertblock}
\end{frame}

\subsection{Conclusion générale}

\begin{frame}{Conclusion générale}
  \semis{south west}{3.2}{2.6}{0.14}{202}%
  \rubrique{Principaux apports}
  \begin{enumerate}
      \item Mesure de la pauvreté multidimensionnelle des enfants sur deux
            vagues EHCVM selon l'approche MODA (7 dimensions, $k=4$).
      \item L'incidence MODA recule de \chiffre{63,5\,\%} (2018) à \chiffre{57,8\,\%} (2021) :
            amélioration réelle mais la pauvreté reste massive.
      \item Les transferts ne réduisent pas, à trois ans, la
            pauvreté multidimensionnelle des enfants : l'effet est de
            \chiffre{$+0{,}064$}, significatif à 5\,\%, de signe défavorable et stable aux
            trois appariements.
      \item Il n'est établi que sur la nutrition, chez les 0-4 ans et sur les
          plus faibles montants reçus, là où l'apport monétaire est trop
          faible pour peser sur les conditions de vie de l'enfant.
  \end{enumerate}
\end{frame}

\subsection{Implications}

\begin{frame}{Implications de politique économique}
  \begin{block}{Cibler les marges où l'écart se concentre}
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item[{\color{presvert}\faLeaf}] Sécurité alimentaire des enfants en
            priorité dans les zones d'émigration.
      \item[{\color{presvert}\faChild}] Appui ciblé aux enfants de 0 à 4 ans
            des ménages bénéficiaires.
      \item[{\color{presvert}\faHandHoldingUsd}] Accompagner en priorité les
            ménages recevant les plus faibles montants.
    \end{itemize}
  \end{block}
  \begin{block}{Améliorer les conditions de conversion des transferts}
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item[{\color{presvert}\faHandHoldingUsd}] Réduire les coûts de transaction
            des envois formels.
      \item[{\color{presvert}\faTint}] Compléter les transferts par l'offre
            publique de services de base : eau, assainissement, santé.
      \item[{\color{presvert}\faBook}] Orienter les usages vers le capital
            humain et renforcer le suivi statistique des transferts.
    \end{itemize}
  \end{block}
\end{frame}
