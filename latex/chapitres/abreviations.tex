\section*{Liste des abréviations}
\addcontentsline{toc}{section}{Liste des abréviations}

\zebre

\begin{longtable}{|l|>{\raggedright\arraybackslash}p{11.6cm}|}
  ANSD    & Agence nationale de la statistique et de la démographie \\ \hline
  ATT     & Average treatment effect on the treated (effet moyen sur les traités) \\ \hline
  DD      & Double différence \\ \hline
  DHS     & Demographic and Health Surveys (enquêtes démographiques et de santé) \\ \hline
  EHCVM   & Enquête harmonisée sur les conditions de vie des ménages \\ \hline
  ENSAE   & École nationale de la statistique et de l'analyse économique \\ \hline
  FCFA    & Franc de la Communauté financière africaine \\ \hline
  FIES    & Food Insecurity Experience Scale (échelle de l'insécurité alimentaire vécue) \\ \hline
  k-NN    & k plus proches voisins (k-nearest neighbors) \\ \hline
  LSMS    & Living Standards Measurement Study \\ \hline
  MAFE    & Migrations entre l'Afrique et l'Europe \\ \hline
  MESE    & Enquête longitudinale sur les migrants sénégalais en Espagne \\ \hline
  MICS    & Multiple Indicator Cluster Surveys (enquêtes par grappes à indicateurs multiples) \\ \hline
  MODA    & Multiple Overlapping Deprivation Analysis (analyse des privations multiples et superposées) \\ \hline
  N-MODA  & MODA adaptée au contexte national \\ \hline
  NEET    & Not in employment, education or training (ni en emploi, ni en études, ni en formation) \\ \hline
  OSIM    & Organisation de solidarité internationale issue de la migration \\ \hline
  PSM     & Propensity score matching (appariement par score de propension) \\ \hline
  PSM-DD  & Appariement par score de propension combiné à la double différence \\ \hline
  SMD     & Standardized mean difference (différence standardisée moyenne) \\ \hline
  UEMOA   & Union économique et monétaire ouest-africaine \\ \hline
  UNICEF  & Fonds des Nations unies pour l'enfance \\ \hline
\end{longtable}
